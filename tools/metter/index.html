const WINDOW_DURATION_MS = 5000; let noiseMaxBuffer = []; // Lưu { time, val } cho âm thanh let lightMaxBuffer = []; // Lưu { time, val } cho ánh sáng /** * Lấy giá trị cực đại (Max) trong cửa sổ trượt 5 giây * @param {Array} buffer Mảng lưu trữ mẫu * @param {number} currentVal Giá trị đo tức thời * @returns {number} Giá trị đỉnh lớn nhất trong 5s vừa qua */ function getMax5s(buffer, currentVal) { const now = performance.now(); // 1. Ghi nhận mẫu tức thời mới nhất buffer.push({ time: now, val: currentVal }); // 2. Xóa bỏ các mẫu đã cũ hơn 5 giây while (buffer.length > 0 && (now - buffer[0].time) > WINDOW_DURATION_MS) { buffer.shift(); } // 3. Tìm giá trị lớn nhất trong 5 giây gần nhất let max = -Infinity; for (let i = 0; i < buffer.length; i++) { if (buffer[i].val > max) { max = buffer[i].val; } } return max !== -Infinity ? max : currentVal; }
